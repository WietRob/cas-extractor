#### Neue Funktionen (H1):

```python
def _build_local_var_types(
    func_node: ast.FunctionDef | ast.AsyncFunctionDef,
    local_classes: dict[str, "ClassInfo"],
    local_imports: dict[str, str],
    all_classes: dict[str, "ClassInfo"],
) -> dict[str, str]:
    """
    H1: Build mapping of local variable names to their inferred class qnames.
    Only tracks: x = ClassName(...) where ClassName is a known class.
    Last assignment wins within the function scope.
    """
    var_types: dict[str, str] = {}
    for node in ast.walk(func_node):
        if isinstance(node, ast.Assign):
            for target in node.targets:
                if isinstance(target, ast.Name):
                    var_name = target.id
                    class_qname = _infer_class_from_expr(
                        node.value, local_classes, local_imports, all_classes
                    )
                    if class_qname:
                        var_types[var_name] = class_qname
    return var_types

def _infer_class_from_expr(
    expr: ast.expr,
    local_classes: dict[str, "ClassInfo"],
    local_imports: dict[str, str],
    all_classes: dict[str, "ClassInfo"],
) -> str | None:
    """H1: Infer class qname from an expression (only simple constructor calls)."""
    if isinstance(expr, ast.Call):
        if isinstance(expr.func, ast.Name):
            class_name = expr.func.id
            if class_name in local_classes:
                return local_classes[class_name].qname
            if class_name in local_imports:
                qualified = local_imports[class_name]
                if qualified in all_classes:
                    return qualified
    return None

def _find_enclosing_func_node(
    node: ast.AST,
) -> ast.FunctionDef | ast.AsyncFunctionDef | None:
    """Find the enclosing function node by walking _parent chain."""
    current = node
    while hasattr(current, "_parent"):
        parent = current._parent
        if isinstance(parent, (ast.FunctionDef, ast.AsyncFunctionDef)):
            return parent
        current = parent
    return None
```

## B) Mini-Fixture Tests

### Test Code (`/tmp/h1-test/test_h1.py`)

```python
class HTTPClient:
    def send(self, data): pass
    def close(self): pass

class ClassA:
    def foo(self): pass

class ClassB:
    def foo(self): pass

class Factory:
    @classmethod
    def create(cls):
        return HTTPClient()

def test_h1_positive():
    """Positive: local var assigned from constructor"""
    client = HTTPClient("api.example.com")
    client.send("hello")  # Should resolve to test_h1.HTTPClient.send
    client.close()        # Should resolve to test_h1.HTTPClient.close

def test_h1_reassignment():
    """Reassignment: last assignment wins"""
    x = ClassA()
    x = ClassB()  # Last assignment wins
    x.foo()       # Should resolve to test_h1.ClassB.foo

def test_h1_multiple_vars():
    """Multiple variables with different types"""
    a = ClassA()
    b = ClassB()
    a.foo()  # ClassA.foo
    b.foo()  # ClassB.foo

def test_h1_negative_factory():
    """Negative: factory pattern - cannot infer type"""
    factory = Factory()
    result = factory.create()  # Factory return - no type info
    result.send("data")  # Should be unresolved

def test_h1_negative_unknown():
    """Negative: unknown class"""
    x = UnknownClass()
    x.foo()  # Should be unresolved
```

**Implementierte Constraints (alle erfüllt):**
- ✅ Nur innerhalb derselben Funktion / desselben Scopes
- ✅ Nur wenn RHS ein Constructor-Call auf bekannte lokale Klasse ist
- ✅ Nur wenn Methode in dieser Klasse existiert
- ✅ Bei Reassignment: last assignment wins
- ✅ Wenn unsicher → unresolved (kein Guessing)
